import fs from "node:fs";
import { chromium } from "playwright";
const dir="docs/proof/stage948";
const records=fs.readdirSync(dir).filter(f=>f.endsWith(".json")).map(f=>JSON.parse(fs.readFileSync(dir+"/"+f))).filter(r=>r.source);
if(records.length!==43||records.some(r=>!r.conditioned||!fs.existsSync(r.file)))throw Error("Incomplete pack");
const browser=await chromium.launch({args:["--no-proxy-server"]});
try{
 const page=await browser.newPage({viewport:{width:1160,height:900}});
 const checked=[];
 for(const r of records){
  const mime=r.file.endsWith(".jpg")?"image/jpeg":"image/png";
  const data="data:"+mime+";base64,"+fs.readFileSync(r.file).toString("base64");
  const stats=await page.evaluate(async src=>{
   const im=new Image();im.src=src;await im.decode();
   const c=document.createElement("canvas");c.width=im.width;c.height=im.height;
   const ctx=c.getContext("2d");ctx.drawImage(im,0,0);
   const d=ctx.getImageData(0,0,c.width,c.height).data;
   const linear=v=>{v/=255;return v<=.04045?v/12.92:((v+.055)/1.055)**2.4;};
   let sum=0,transparent=0;for(let i=0;i<d.length;i+=4){sum+=.2126*linear(d[i])+.7152*linear(d[i+1])+.0722*linear(d[i+2]);if(d[i+3]<128)transparent++;}
   return {width:im.width,height:im.height,linearLuma:sum/(d.length/4),transparentFraction:transparent/(d.length/4)};
  },data);
  checked.push({id:r.id,...stats});
 }
 fs.writeFileSync(dir+"/decoded-images.json",JSON.stringify(checked,null,2)+"\n");
 for(const [group,list,per] of [["textures",records.filter(r=>r.id.startsWith("tex_")),10],["arrivals",records.filter(r=>r.kind==="district"),4],["portraits",records.filter(r=>r.kind==="portrait"),5]]){
  for(let off=0;off<list.length;off+=per){
   const slice=list.slice(off,off+per);
   await page.setContent('<style>body{margin:20px;background:#0c1119;color:#b5ccd5;font:14px monospace}main{display:grid;grid-template-columns:repeat('+(group==="arrivals"?2:5)+',1fr);gap:14px}img{width:100%;height:auto;background:#283442}label{display:block;margin:7px 0;font-size:12px}h1{font-size:20px}</style><h1>MELTDOWN / '+group+'</h1><main>'+slice.map(r=>'<section><img src="data:'+(r.file.endsWith(".jpg")?"image/jpeg":"image/png")+';base64,'+fs.readFileSync(r.file).toString("base64")+'"><label>'+r.id+'</label></section>').join("")+"</main>");
   await page.evaluate(()=>Promise.all([...document.images].map(im=>im.decode())));
   await page.screenshot({path:dir+"/"+group+"-"+(Math.floor(off/per)+1)+".png",fullPage:true});
  }
 }
 console.log(JSON.stringify({decoded:checked.length,slab:checked.find(r=>r.id==="tex_plaza_slab"),chain:checked.find(r=>r.id==="tex_chain_woven")},null,2));
}finally{await browser.close();}
