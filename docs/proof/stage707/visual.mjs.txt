import { chromium } from 'playwright';
const browser = await chromium.launch({args:['--no-proxy-server','--use-angle=metal','--use-gl=angle','--ignore-gpu-blocklist']});
const errors=[];
try {
 const page = await browser.newPage({viewport:{width:960,height:600}});
 page.on('pageerror',e=>errors.push(String(e)));
 await page.goto('http://127.0.0.1:5234/probe/out/art707.html',{waitUntil:'domcontentloaded',timeout:120000});
 await page.waitForFunction(()=>window.review && document.querySelector('#loading .art')?.naturalWidth===960,null,{timeout:60000});
 await page.screenshot({timeout:120000,animations:"disabled",path:'docs/proof/stage707/lease-row-loading.png'});
 console.log("Loading-card screenshot captured");
 const loading=await page.evaluate(()=>window.review.card.view());
 await page.evaluate(()=>{window.review.card.hide(false);window.review.preview.start();});
 await page.waitForFunction(()=>window.review.preview.rig.mat.map?.image?.src.endsWith('/tex_cloak_v2.png') && window.review.preview.renderer?.info.render.calls>0,null,{timeout:60000});
 await page.waitForTimeout(1000);
 await page.evaluate(()=>window.review.preview.stop());
 await page.screenshot({timeout:120000,animations:"disabled",path:'docs/proof/stage707/character-cloth.png'});
 const cloth=await page.evaluate(()=>({source:window.review.preview.rig.mat.map.image.src.split('/').pop(),width:window.review.preview.rig.mat.map.image.width,calls:window.review.preview.renderer.info.render.calls,assets:window.review.assetStats}));
 console.log(JSON.stringify({loading,cloth,errors},null,2));
 if (loading.art!=='/districts/lease_row.jpg'||cloth.width!==256||cloth.assets.failed||errors.length) process.exitCode=1;
} finally {await browser.close();}
